deploy_new_client() {
  CLIENT_ID=$1
  PORT=$2
  DOMAIN=$3
  ENV_CONTENT=$4

  # Crear directorios de uploads exclusivos para este cliente
  mkdir -p /root/backend-qhubomor/public/uploads/$CLIENT_ID/insumos
  mkdir -p /root/backend-qhubomor/public/uploads/$CLIENT_ID/gastos
  mkdir -p /root/backend-qhubomor/public/uploads/$CLIENT_ID/asistencia
  mkdir -p /root/backend-qhubomor/public/uploads/$CLIENT_ID/productos

  # Detener y eliminar contenedor anterior de este cliente
  docker stop backend-$CLIENT_ID || true
  docker rm backend-$CLIENT_ID || true

  # Sincronizar esquema en la BD de este cliente (sin borrar datos)
  echo "Sincronizando BD para $CLIENT_ID..."
  docker run --rm \
    --env-file /root/backend-qhubomor/.env.$CLIENT_ID \
    qhubomor-backend \
    npx prisma db push --accept-data-loss

  # Levantar contenedor de este cliente (usa la imagen ya construida arriba)
  docker run -d \
    --name backend-$CLIENT_ID \
    -p $PORT:3000 \
    -v /root/backend-qhubomor/public/uploads/$CLIENT_ID:/app/public/uploads \
    --restart unless-stopped \
    --env-file /root/backend-qhubomor/.env.$CLIENT_ID \
    qhubomor-backend

  # Configurar subdominio + certificado SSL automáticamente usando el script de n8n
  DOMAIN=$DOMAIN PORT=$PORT EMAIL=jonnymejia62@gmail.com \
    bash /root/backend-qhubomor/scripts/setup-nginx.sh

  echo "✅ Cliente $CLIENT_ID desplegado en https://$DOMAIN"
}
